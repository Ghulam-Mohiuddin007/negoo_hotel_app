🎯 Dart Realtime Stream Listener lib/live_bids_tab.dart 
// Stream active bids directly for this hotel
StreamBuilder<List<Map<String, dynamic>>>(
  stream: Supabase.instance.client
      .from('bids')
      .stream(primaryKey: ['id'])
      .eq('hotel_id', widget.currentHotelId),
  builder: (context, snapshot) {
    final liveBids = (snapshot.data ?? [])
        .where((b) => b['status'] == 'pending')
        .toList();

    // Trigger physical haptic alert on new bid
    if (liveBids.length > _lastBidCount) {
      HapticFeedback.vibrate();
    }
    return ListView.builder(...);
  },
)
